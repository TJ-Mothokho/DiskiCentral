import { useParams, Link } from "react-router-dom";
import { Clock, Calendar, Eye, Share2, ArrowRight, Mail } from "lucide-react";
import { useState } from "react";
import CategoryBadge from "../components/CategoryBadge";
import ArticleCard from "../components/ArticleCard";

import articles from "../data/articles.json";

interface ArticleProps { darkMode: boolean }

export default function Article({ darkMode }: ArticleProps) {
  const { slug } = useParams<{ slug: string }>();
  const [email, setEmail] = useState("");
  const [subscribed, setSubscribed] = useState(false);

  const article = articles.find((a) => a.slug === slug);
  if (!article) {
    return (
      <div className="max-w-[1440px] mx-auto px-4 py-24 text-center">
        <h1 className="font-display font-bold text-4xl mb-4">Article Not Found</h1>
        <Link to="/news" className="text-[#00C853] font-semibold">← Back to News</Link>
      </div>
    );
  }

  const related = articles.filter((a) => article.relatedArticles.includes(a.id)).slice(0, 3);
  const moreFromCategory = articles
    .filter((a) => a.categorySlug === article.categorySlug && a.id !== article.id)
    .slice(0, 3);

  return (
    <main className="max-w-[1440px] mx-auto px-4 py-6">
      <div className="flex flex-col lg:flex-row gap-8">
        {/* Article content */}
        <article className="flex-1 min-w-0 max-w-3xl">
          {/* Breadcrumb */}
          <div className={`flex items-center gap-2 text-xs mb-5 ${darkMode ? "text-gray-400" : "text-gray-500"}`}>
            <Link to="/" className="hover:text-[#00C853] transition-colors">Home</Link>
            <span>/</span>
            <Link to="/news" className="hover:text-[#00C853] transition-colors">News</Link>
            <span>/</span>
            <span className="truncate">{article.title}</span>
          </div>

          {/* Meta */}
          <div className="flex items-center gap-2 mb-4">
            <CategoryBadge category={article.category} categorySlug={article.categorySlug} />
            {article.trending && (
              <span className="bg-orange-500 text-white text-xs font-bold px-2 py-0.5 rounded-full uppercase tracking-wide">
                Trending
              </span>
            )}
          </div>

          {/* Headline */}
          <h1 className={`font-display font-bold text-3xl sm:text-4xl leading-tight mb-3 ${darkMode ? "text-white" : "text-gray-900"}`}>
            {article.title}
          </h1>
          <p className={`text-lg leading-relaxed mb-5 ${darkMode ? "text-gray-300" : "text-gray-600"}`}>
            {article.subtitle}
          </p>

          {/* Author & meta */}
          <div className={`flex flex-wrap items-center gap-4 pb-5 mb-5 border-b ${darkMode ? "border-gray-800" : "border-gray-100"}`}>
            <div className="flex items-center gap-2.5">
              <img
                src={article.authorAvatar}
                alt={article.author}
                className="w-9 h-9 rounded-full object-cover"
              />
              <div>
                <div className={`text-sm font-semibold ${darkMode ? "text-gray-200" : "text-gray-800"}`}>
                  {article.author}
                </div>
              </div>
            </div>
            <div className={`flex items-center gap-4 text-xs ${darkMode ? "text-gray-400" : "text-gray-500"}`}>
              <span className="flex items-center gap-1">
                <Calendar size={12} />
                {new Date(article.date).toLocaleDateString("en-ZA", { day: "numeric", month: "long", year: "numeric" })}
              </span>
              <span className="flex items-center gap-1">
                <Clock size={12} />
                {article.readingTime} min read
              </span>
              {article.views && (
                <span className="flex items-center gap-1">
                  <Eye size={12} />
                  {article.views.toLocaleString()} views
                </span>
              )}
            </div>

            {/* Share */}
            <div className="flex items-center gap-2 ml-auto">
              <span className={`text-xs font-medium ${darkMode ? "text-gray-400" : "text-gray-500"}`}>Share:</span>
              <button className="p-1.5 rounded-full bg-blue-500 text-white hover:bg-blue-600 transition-colors text-xs font-bold px-2">
                𝕏
              </button>
              <button className="p-1.5 rounded-full bg-blue-700 text-white hover:bg-blue-800 transition-colors text-xs font-bold px-2">
                fb
              </button>
              <button className={`p-1.5 rounded-full transition-colors ${darkMode ? "bg-gray-700 text-gray-300 hover:bg-gray-600" : "bg-gray-100 text-gray-600 hover:bg-gray-200"}`}>
                <Share2 size={13} />
              </button>
            </div>
          </div>

          {/* Hero image */}
          <div className="rounded-xl overflow-hidden mb-6 bg-gray-200 aspect-[16/9]">
            <img
              src={article.heroImage}
              alt={article.title}
              className="w-full h-full object-cover"
            />
          </div>

          {/* Ad placeholder */}
          <div className={`rounded-lg border-2 border-dashed flex items-center justify-center h-16 mb-6 text-xs ${darkMode ? "border-gray-700 text-gray-600" : "border-gray-200 text-gray-400"}`}>
            Advertisement
          </div>

          {/* Body */}
          <div
            className={`prose prose-base max-w-none mb-8 ${darkMode ? "prose-invert" : ""}`}
            style={{
              lineHeight: "1.8",
            }}
          >
            <div
              className={`leading-relaxed space-y-4 ${darkMode ? "text-gray-200" : "text-gray-800"}`}
              dangerouslySetInnerHTML={{
                __html: article.body
                  .replace(/<blockquote>/g, `<blockquote class="border-l-4 border-[#00C853] pl-4 my-6 italic ${darkMode ? "text-gray-300" : "text-gray-600"} text-lg">`)
                  .replace(/<p>/g, `<p class="mb-4 text-base leading-relaxed">`)
              }}
            />
          </div>

          {/* Tags */}
          <div className={`flex flex-wrap items-center gap-2 pb-6 border-b ${darkMode ? "border-gray-800" : "border-gray-100"}`}>
            <span className={`text-xs font-semibold uppercase tracking-wide ${darkMode ? "text-gray-400" : "text-gray-500"}`}>
              Tags:
            </span>
            {article.tags.map((tag) => (
              <Link
                key={tag}
                to={`/search?q=${encodeURIComponent(tag)}`}
                className={`px-2.5 py-1 rounded-full text-xs font-medium transition-colors hover:bg-[#00C853] hover:text-black ${darkMode ? "bg-gray-800 text-gray-300" : "bg-gray-100 text-gray-600"}`}
              >
                {tag}
              </Link>
            ))}
          </div>

          {/* Author card */}
          <div className={`rounded-xl border p-5 my-6 ${darkMode ? "bg-gray-900 border-gray-800" : "bg-gray-50 border-gray-100"}`}>
            <div className="flex items-center gap-4">
              <img
                src={article.authorAvatar}
                alt={article.author}
                className="w-14 h-14 rounded-full object-cover"
              />
              <div>
                <h3 className={`font-display font-bold text-base ${darkMode ? "text-white" : "text-gray-900"}`}>{article.author}</h3>
                <p className={`text-xs mt-0.5 ${darkMode ? "text-gray-400" : "text-gray-500"}`}>
                  Football journalist at DiskiCentral
                </p>
              </div>
            </div>
          </div>

          {/* Newsletter CTA */}
          <div className="bg-[#0A0A0A] rounded-xl p-6 my-6 text-center">
            <h3 className="font-display font-bold text-xl text-white mb-1">Enjoy this article?</h3>
            <p className="text-gray-400 text-sm mb-4">Subscribe for daily South African football coverage.</p>
            {subscribed ? (
              <span className="text-[#00C853] text-sm font-semibold">✓ Subscribed!</span>
            ) : (
              <form onSubmit={(e) => { e.preventDefault(); if (email) setSubscribed(true); }} className="flex gap-2 max-w-sm mx-auto">
                <input
                  type="email"
                  value={email}
                  onChange={(e) => setEmail(e.target.value)}
                  placeholder="Your email"
                  className="flex-1 px-3 py-2 bg-gray-800 border border-gray-700 text-white rounded-lg text-sm outline-none focus:border-[#00C853] transition-colors placeholder-gray-500"
                  required
                />
                <button type="submit" className="px-4 py-2 bg-[#00C853] text-black font-bold text-sm rounded-lg hover:bg-[#00A344] transition-colors">
                  Subscribe
                </button>
              </form>
            )}
          </div>

          {/* Related Articles */}
          {related.length > 0 && (
            <section className="my-8">
              <h2 className={`font-display font-bold text-xl mb-4 ${darkMode ? "text-white" : "text-gray-900"}`}>
                Related Articles
              </h2>
              <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
                {related.map((a) => (
                  <ArticleCard key={a.id} article={a} darkMode={darkMode} />
                ))}
              </div>
            </section>
          )}

          {/* More from category */}
          {moreFromCategory.length > 0 && (
            <section className="my-8">
              <h2 className={`font-display font-bold text-xl mb-4 ${darkMode ? "text-white" : "text-gray-900"}`}>
                More from {article.category}
              </h2>
              <div className="space-y-3">
                {moreFromCategory.map((a) => (
                  <ArticleCard key={a.id} article={a} variant="horizontal" darkMode={darkMode} />
                ))}
              </div>
            </section>
          )}
        </article>

        {/* Sidebar */}
        <aside className="w-full lg:w-72 shrink-0 space-y-5">
          {/* Ad */}
          <div className={`rounded-xl border-2 border-dashed flex items-center justify-center h-48 text-xs ${darkMode ? "border-gray-700 text-gray-600" : "border-gray-200 text-gray-400"}`}>
            Advertisement 300×250
          </div>

          {/* Latest articles */}
          <div className={`rounded-xl border p-4 ${darkMode ? "bg-gray-900 border-gray-800" : "bg-white border-gray-100"}`}>
            <h3 className={`font-display font-bold text-sm uppercase tracking-wide mb-4 ${darkMode ? "text-white" : "text-gray-900"}`}>
              Latest News
            </h3>
            <div className="space-y-4">
              {articles.filter((a) => a.id !== article.id).slice(0, 5).map((a) => (
                <ArticleCard key={a.id} article={a} variant="horizontal" darkMode={darkMode} />
              ))}
            </div>
          </div>

          {/* Ad */}
          <div className={`rounded-xl border-2 border-dashed flex items-center justify-center h-48 text-xs ${darkMode ? "border-gray-700 text-gray-600" : "border-gray-200 text-gray-400"}`}>
            Advertisement 300×250
          </div>
        </aside>
      </div>
    </main>
  );
}
