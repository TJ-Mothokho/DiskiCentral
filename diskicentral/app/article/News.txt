import { useState, useMemo } from "react";
import { Search, Filter, ChevronLeft, ChevronRight, TrendingUp, Play, Tag } from "lucide-react";
import ArticleCard from "../components/ArticleCard";
import CategoryBadge from "../components/CategoryBadge";
import VideoCard from "../components/VideoCard";

import articles from "../data/articles.json";
import categories from "../data/categories.json";
import videos from "../data/videos.json";

interface NewsProps { darkMode: boolean }

const ARTICLES_PER_PAGE = 6;

export default function News({ darkMode }: NewsProps) {
  const [search, setSearch] = useState("");
  const [selectedCategory, setSelectedCategory] = useState<string | null>(null);
  const [page, setPage] = useState(1);

  const filtered = useMemo(() => {
    let result = [...articles];
    if (selectedCategory) result = result.filter((a) => a.categorySlug === selectedCategory);
    if (search.trim()) {
      const q = search.toLowerCase();
      result = result.filter(
        (a) =>
          a.title.toLowerCase().includes(q) ||
          a.excerpt.toLowerCase().includes(q) ||
          a.author.toLowerCase().includes(q)
      );
    }
    return result;
  }, [search, selectedCategory]);

  const totalPages = Math.ceil(filtered.length / ARTICLES_PER_PAGE);
  const paginated = filtered.slice((page - 1) * ARTICLES_PER_PAGE, page * ARTICLES_PER_PAGE);
  const trending = articles.filter((a) => a.trending).slice(0, 5);
  const popularTags = ["PSL", "Bafana Bafana", "Sundowns", "Pirates", "Chiefs", "AFCON", "Transfers", "CAF"];

  return (
    <main className="max-w-[1440px] mx-auto px-4 py-6">
      <div className="mb-8">
        <h1 className={`font-display font-bold text-4xl mb-1 ${darkMode ? "text-white" : "text-gray-900"}`}>News</h1>
        <p className={`text-base ${darkMode ? "text-gray-400" : "text-gray-500"}`}>
          All the latest from South African football
        </p>
      </div>

      <div className="flex flex-col lg:flex-row gap-8">
        {/* Main content */}
        <div className="flex-1 min-w-0">
          {/* Search & Filters */}
          <div className="flex flex-col sm:flex-row gap-3 mb-5">
            <div className="relative flex-1">
              <Search size={16} className="absolute left-3 top-1/2 -translate-y-1/2 text-gray-400" />
              <input
                type="text"
                value={search}
                onChange={(e) => { setSearch(e.target.value); setPage(1); }}
                placeholder="Search articles, authors..."
                className={`w-full pl-9 pr-4 py-2.5 text-sm border rounded-xl outline-none focus:border-[#00C853] transition-colors ${darkMode ? "bg-gray-800 border-gray-700 text-white placeholder-gray-500" : "bg-white border-gray-200 text-gray-900"}`}
              />
            </div>
            <div className={`flex items-center gap-1.5 px-3 py-2.5 rounded-xl border text-sm ${darkMode ? "border-gray-700 text-gray-400 bg-gray-800" : "border-gray-200 text-gray-500 bg-white"}`}>
              <Filter size={14} />
              <span>Filter</span>
            </div>
          </div>

          {/* Category chips */}
          <div className="flex flex-wrap gap-2 mb-6">
            <button
              onClick={() => { setSelectedCategory(null); setPage(1); }}
              className={`px-3 py-1.5 rounded-full text-xs font-semibold transition-colors ${
                !selectedCategory
                  ? "bg-[#0A0A0A] text-white"
                  : darkMode ? "bg-gray-800 text-gray-300 hover:bg-gray-700" : "bg-gray-100 text-gray-600 hover:bg-gray-200"
              }`}
            >
              All
            </button>
            {categories.map((cat) => (
              <button
                key={cat.id}
                onClick={() => { setSelectedCategory(cat.slug === selectedCategory ? null : cat.slug); setPage(1); }}
                className={`px-3 py-1.5 rounded-full text-xs font-semibold transition-colors ${
                  selectedCategory === cat.slug
                    ? "bg-[#00C853] text-black"
                    : darkMode ? "bg-gray-800 text-gray-300 hover:bg-gray-700" : "bg-gray-100 text-gray-600 hover:bg-gray-200"
                }`}
              >
                {cat.name}
              </button>
            ))}
          </div>

          {/* Results count */}
          <p className={`text-sm mb-4 ${darkMode ? "text-gray-400" : "text-gray-500"}`}>
            {filtered.length} article{filtered.length !== 1 ? "s" : ""}
            {selectedCategory && ` in ${categories.find((c) => c.slug === selectedCategory)?.name}`}
            {search && ` matching "${search}"`}
          </p>

          {/* Article Grid */}
          {paginated.length > 0 ? (
            <div className="grid grid-cols-1 sm:grid-cols-2 gap-5 mb-6">
              {paginated.map((article) => (
                <ArticleCard key={article.id} article={article} darkMode={darkMode} />
              ))}
            </div>
          ) : (
            <div className={`text-center py-16 rounded-xl border ${darkMode ? "border-gray-800 text-gray-400" : "border-gray-100 text-gray-500"}`}>
              <p className="text-2xl mb-2">⚽</p>
              <p className="font-semibold">No articles found</p>
              <p className="text-sm mt-1">Try a different search term or category</p>
            </div>
          )}

          {/* Pagination */}
          {totalPages > 1 && (
            <div className="flex items-center justify-center gap-2">
              <button
                onClick={() => setPage((p) => Math.max(1, p - 1))}
                disabled={page === 1}
                className={`p-2 rounded-lg transition-colors disabled:opacity-40 ${darkMode ? "bg-gray-800 text-gray-300 hover:bg-gray-700" : "bg-white border border-gray-200 text-gray-600 hover:bg-gray-50"}`}
              >
                <ChevronLeft size={16} />
              </button>
              {Array.from({ length: totalPages }, (_, i) => i + 1).map((p) => (
                <button
                  key={p}
                  onClick={() => setPage(p)}
                  className={`w-9 h-9 rounded-lg text-sm font-semibold transition-colors ${
                    p === page
                      ? "bg-[#00C853] text-black"
                      : darkMode ? "bg-gray-800 text-gray-300 hover:bg-gray-700" : "bg-white border border-gray-200 text-gray-600 hover:bg-gray-50"
                  }`}
                >
                  {p}
                </button>
              ))}
              <button
                onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
                disabled={page === totalPages}
                className={`p-2 rounded-lg transition-colors disabled:opacity-40 ${darkMode ? "bg-gray-800 text-gray-300 hover:bg-gray-700" : "bg-white border border-gray-200 text-gray-600 hover:bg-gray-50"}`}
              >
                <ChevronRight size={16} />
              </button>
            </div>
          )}
        </div>

        {/* Sidebar */}
        <aside className="w-full lg:w-72 shrink-0 space-y-6">
          {/* Trending */}
          <div className={`rounded-xl border p-4 ${darkMode ? "bg-gray-900 border-gray-800" : "bg-white border-gray-100"}`}>
            <div className="flex items-center gap-2 mb-4">
              <TrendingUp size={15} className="text-[#00C853]" />
              <h3 className={`font-display font-bold text-sm uppercase tracking-wide ${darkMode ? "text-white" : "text-gray-900"}`}>
                Trending
              </h3>
            </div>
            <div className="space-y-3">
              {trending.map((article, i) => (
                <div key={article.id} className="flex gap-3 items-start">
                  <span className="font-display font-bold text-xl text-[#00C853]/30 shrink-0 leading-none">{i + 1}</span>
                  <div>
                    <a href={`/article/${article.slug}`} className={`text-xs font-semibold leading-snug hover:text-[#00C853] transition-colors line-clamp-2 ${darkMode ? "text-gray-200" : "text-gray-800"}`}>
                      {article.title}
                    </a>
                    <div className="mt-0.5">
                      <CategoryBadge category={article.category} categorySlug={article.categorySlug} small />
                    </div>
                  </div>
                </div>
              ))}
            </div>
          </div>

          {/* Latest Videos */}
          <div className={`rounded-xl border p-4 ${darkMode ? "bg-gray-900 border-gray-800" : "bg-white border-gray-100"}`}>
            <div className="flex items-center gap-2 mb-4">
              <Play size={14} className="text-[#00C853]" />
              <h3 className={`font-display font-bold text-sm uppercase tracking-wide ${darkMode ? "text-white" : "text-gray-900"}`}>
                Latest Videos
              </h3>
            </div>
            <div className="space-y-3">
              {videos.slice(0, 3).map((v) => (
                <VideoCard key={v.id} video={v} darkMode={darkMode} />
              ))}
            </div>
          </div>

          {/* Popular Tags */}
          <div className={`rounded-xl border p-4 ${darkMode ? "bg-gray-900 border-gray-800" : "bg-white border-gray-100"}`}>
            <div className="flex items-center gap-2 mb-4">
              <Tag size={14} className="text-[#00C853]" />
              <h3 className={`font-display font-bold text-sm uppercase tracking-wide ${darkMode ? "text-white" : "text-gray-900"}`}>
                Popular Tags
              </h3>
            </div>
            <div className="flex flex-wrap gap-2">
              {popularTags.map((tag) => (
                <span key={tag} className={`px-2.5 py-1 rounded-full text-xs font-medium cursor-pointer transition-colors hover:bg-[#00C853] hover:text-black ${darkMode ? "bg-gray-800 text-gray-300" : "bg-gray-100 text-gray-600"}`}>
                  {tag}
                </span>
              ))}
            </div>
          </div>
        </aside>
      </div>
    </main>
  );
}
