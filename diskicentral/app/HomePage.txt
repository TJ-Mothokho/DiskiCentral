import { Link } from "react-router-dom";
import { ArrowRight, Clock, TrendingUp, Play, Mail } from "lucide-react";
import { useState } from "react";
import ArticleCard from "../components/ArticleCard";
import VideoCard from "../components/VideoCard";
import MatchCard from "../components/MatchCard";
import CategoryBadge from "../components/CategoryBadge";

import articles from "../data/articles.json";
import videos from "../data/videos.json";
import fixtures from "../data/fixtures.json";
import results from "../data/results.json";
import players from "../data/players.json";
import transfers from "../data/transfers.json";

  const [subscribed, setSubscribed] = useState(false);

  const latest = articles.slice(1, 7);

  const cafArticles = articles.filter((a) => a.categorySlug === "caf").slice(0, 3);
  const opinionArticles = articles.filter((a) => a.categorySlug === "opinion").slice(0, 2);

  const upcomingFixtures = fixtures.slice(0, 3);
  const recentResults = results.slice(0, 3);
  const featuredVideo = videos[0];
  const latestVideos = videos.slice(1, 4);

          {/* Main hero */}
          <Link
            to={`/article/${featured.slug}`}
            className="group lg:col-span-2 relative rounded-2xl overflow-hidden block aspect-[16/10] bg-gray-900"
          >
            <img
              src={featured.heroImage}
              alt={featured.title}
              className="w-full h-full object-cover opacity-70 group-hover:opacity-80 group-hover:scale-105 transition-all duration-700"

                <span className="text-white/70 text-xs flex items-center gap-1">
                  <Clock size={11} /> {featured.readingTime} min read
                </span>
              </div>
              <h1 className="font-display font-bold text-2xl sm:text-3xl lg:text-4xl text-white leading-tight mb-3 group-hover:text-[#00C853] transition-colors">
                {featured.title}
              </h1>
              <p className="text-gray-300 text-sm sm:text-base leading-relaxed mb-4 line-clamp-2 max-w-2xl">
                {featured.excerpt}
              </p>
              <div className="flex items-center justify-between">
                <div className="flex items-center gap-2 text-sm text-gray-300">
                  <span className="font-medium text-white">{featured.author}</span>
                  <span>·</span>
                  <span>{new Date(featured.date).toLocaleDateString("en-ZA", { day: "numeric", month: "short", year: "numeric" })}</span>
                </div>
                <span className="flex items-center gap-1.5 text-[#00C853] text-sm font-semibold">
                  Read More <ArrowRight size={14} />
                </span>
              </div>
            </div>
          </Link>

          {/* Side trending */}
          <div className={`rounded-2xl border p-5 ${darkMode ? "bg-gray-900 border-gray-800" : "bg-white border-gray-100"}`}>
            <div className="flex items-center gap-2 mb-4">
              <TrendingUp size={16} className="text-[#00C853]" />
              <h2 className={`font-display font-bold text-lg uppercase tracking-wide ${darkMode ? "text-white" : "text-gray-900"}`}>
                Trending
              </h2>
            </div>
            <div className="space-y-4">
              {trending.map((article, i) => (
                <div key={article.id} className="flex gap-3">
                  <span className="font-display font-bold text-3xl leading-none text-[#00C853]/30 shrink-0 w-8">
                    {String(i + 1).padStart(2, "0")}
                  </span>
                  <div>
                    <Link
                      to={`/article/${article.slug}`}
                      className={`text-sm font-semibold leading-snug hover:text-[#00C853] transition-colors line-clamp-2 ${darkMode ? "text-gray-200" : "text-gray-900"}`}
                    >
                      {article.title}
                    </Link>
                    <div className={`flex items-center gap-1.5 mt-1 text-xs ${darkMode ? "text-gray-500" : "text-gray-400"}`}>
                      <CategoryBadge category={article.category} categorySlug={article.categorySlug} small />
                      <span>{new Date(article.date).toLocaleDateString("en-ZA", { day: "numeric", month: "short" })}</span>
                    </div>
                  </div>
                </div>
              ))}
            </div>
          </div>
        </div>
      </section>

      {/* Latest News */}
      <section className="max-w-[1440px] mx-auto px-4 py-6">
        <div className="flex items-center justify-between mb-5">
          <h2 className={`font-display font-bold text-2xl uppercase tracking-wide ${darkMode ? "text-white" : "text-gray-900"}`}>
            Latest News
          </h2>
          <Link to="/news" className="flex items-center gap-1 text-sm text-[#00C853] font-semibold hover:gap-2 transition-all">
            All News <ArrowRight size={14} />
          </Link>
        </div>
        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-5">
          {latest.map((article) => (
            <ArticleCard key={article.id} article={article} darkMode={darkMode} />
          ))}
        </div>
      </section>

      {/* Match Centre */}
      <section className={`border-y py-8 ${darkMode ? "border-gray-800 bg-gray-950" : "border-gray-100 bg-gray-50"}`}>
        <div className="max-w-[1440px] mx-auto px-4">
          <div className="flex items-center justify-between mb-5">
            <h2 className={`font-display font-bold text-2xl uppercase tracking-wide ${darkMode ? "text-white" : "text-gray-900"}`}>
              Match Centre
            </h2>
          </div>
          <div className="grid grid-cols-1 lg:grid-cols-2 gap-8">
            <div>
              <h3 className={`text-sm font-semibold uppercase tracking-widest mb-3 ${darkMode ? "text-gray-400" : "text-gray-500"}`}>
                Upcoming Fixtures
              </h3>
              <div className="space-y-3">
                {upcomingFixtures.map((f) => (
                  <MatchCard key={f.id} match={f} darkMode={darkMode} />
                ))}
              </div>
            </div>
            <div>
              <h3 className={`text-sm font-semibold uppercase tracking-widest mb-3 ${darkMode ? "text-gray-400" : "text-gray-500"}`}>
                Recent Results
              </h3>
              <div className="space-y-3">
                {recentResults.map((r) => (
                  <MatchCard key={r.id} match={{ ...r, status: "completed" }} darkMode={darkMode} />
                ))}
              </div>
            </div>
          </div>
        </div>
      </section>

      {/* PSL Section */}
      <section className="max-w-[1440px] mx-auto px-4 py-8">
        <div className="flex items-center justify-between mb-5">
          <div className="flex items-center gap-3">
            <div className="w-1 h-6 bg-[#00C853] rounded-full" />
            <h2 className={`font-display font-bold text-2xl uppercase tracking-wide ${darkMode ? "text-white" : "text-gray-900"}`}>
              PSL
            </h2>
            <span className={`text-sm font-medium ${darkMode ? "text-gray-400" : "text-gray-500"}`}>DStv Premiership</span>

          <Link to="/competitions/psl" className="flex items-center gap-1 text-sm text-[#00C853] font-semibold hover:gap-2 transition-all">
            More PSL <ArrowRight size={14} />
          </Link>
        </div>
        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-5">
          {pslArticles.map((article) => (
            <ArticleCard key={article.id} article={article} darkMode={darkMode} />
          ))}
        </div>
      </section>

      {/* CAF Section */}

        <div className="max-w-[1440px] mx-auto px-4">
          <div className="flex items-center justify-between mb-5">
            <div className="flex items-center gap-3">
              <div className="w-1 h-6 bg-blue-500 rounded-full" />
              <h2 className={`font-display font-bold text-2xl uppercase tracking-wide ${darkMode ? "text-white" : "text-gray-900"}`}>
                CAF
              </h2>

            </Link>
          </div>

            {cafArticles.map((article) => (
              <ArticleCard key={article.id} article={article} darkMode={darkMode} />
            ))}
          </div>
        </div>
      </section>

      {/* South Africans Abroad */}
      <section className="max-w-[1440px] mx-auto px-4 py-8">
        <div className="flex items-center justify-between mb-5">
          <div className="flex items-center gap-3">
            <div className="w-1 h-6 bg-purple-500 rounded-full" />
            <h2 className={`font-display font-bold text-2xl uppercase tracking-wide ${darkMode ? "text-white" : "text-gray-900"}`}>
              South Africans Abroad
            </h2>
          </div>
          <Link to="/players-abroad" className="flex items-center gap-1 text-sm text-purple-500 font-semibold">
            All Players <ArrowRight size={14} />
          </Link>
        </div>
        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
          {abroadPlayers.map((player) => (
            <Link
              key={player.id}
              to={`/player/${player.slug}`}
              className={`group rounded-xl border p-4 hover:border-[#00C853]/30 hover:shadow-md transition-all ${darkMode ? "bg-gray-900 border-gray-800" : "bg-white border-gray-100"}`}
            >
              <div className="flex items-center gap-3 mb-3">
                <div className="w-12 h-12 rounded-full overflow-hidden bg-gray-200 shrink-0">
                  <img src={player.photo} alt={player.name} className="w-full h-full object-cover" />
                </div>
                <div>
                  <h3 className={`font-display font-bold text-sm group-hover:text-[#00C853] transition-colors ${darkMode ? "text-gray-100" : "text-gray-900"}`}>
                    {player.name}
                  </h3>
                  <p className={`text-xs ${darkMode ? "text-gray-500" : "text-gray-400"}`}>{player.position}</p>
                </div>
              </div>
              <div className={`text-xs space-y-1 ${darkMode ? "text-gray-400" : "text-gray-500"}`}>
                <div className="flex justify-between">
                  <span>Club</span>
                  <span className="font-medium text-right">{player.currentClub}</span>
                </div>
                <div className="flex justify-between">
                  <span>League</span>

                </div>
                <div className="flex justify-between">
                  <span>Country</span>
                  <span className="font-medium">{player.currentClubCountry}</span>
                </div>
              </div>
              <div className="mt-3 pt-3 border-t border-gray-100">
                <div className="flex justify-between text-xs">
                  <span className={darkMode ? "text-gray-500" : "text-gray-400"}>Season goals</span>
                  <span className="font-bold text-[#00C853]">{player.stats.goals}</span>
                </div>
              </div>
            </Link>
          ))}
        </div>
      </section>

      {/* Transfer Centre Snapshot */}
      <section className={`border-y py-8 ${darkMode ? "border-gray-800 bg-gray-950" : "border-gray-100 bg-gray-50"}`}>
        <div className="max-w-[1440px] mx-auto px-4">
          <div className="flex items-center justify-between mb-5">
            <div className="flex items-center gap-3">
              <div className="w-1 h-6 bg-orange-500 rounded-full" />
              <h2 className={`font-display font-bold text-2xl uppercase tracking-wide ${darkMode ? "text-white" : "text-gray-900"}`}>
                Transfer Centre
              </h2>
            </div>
            <Link to="/transfers" className="flex items-center gap-1 text-sm text-orange-500 font-semibold">
              All Transfers <ArrowRight size={14} />
            </Link>
          </div>
          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
            {recentTransfers.map((transfer) => (
              <div
                key={transfer.id}
                className={`flex items-center gap-4 p-4 rounded-xl border ${darkMode ? "bg-gray-900 border-gray-800" : "bg-white border-gray-100"}`}
              >
                <div className="w-10 h-10 rounded-full overflow-hidden bg-gray-200 shrink-0">
                  <img src={transfer.playerPhoto} alt={transfer.player} className="w-full h-full object-cover" />
                </div>
                <div className="flex-1 min-w-0">
                  <div className="flex items-center gap-2 mb-0.5">
                    <span className={`font-semibold text-sm ${darkMode ? "text-gray-100" : "text-gray-900"}`}>{transfer.player}</span>
                    <span className={`text-xs px-1.5 py-0.5 rounded-full font-semibold ${
                      transfer.type === "official" ? "bg-green-100 text-green-700" :
                      transfer.type === "rumour" ? "bg-yellow-100 text-yellow-700" :
                      transfer.type === "loan" ? "bg-blue-100 text-blue-700" :
                      "bg-red-100 text-red-700"
                    }`}>
                      {transfer.type}
                    </span>
                  </div>
                  <p className={`text-xs ${darkMode ? "text-gray-400" : "text-gray-500"}`}>
                    {transfer.fromClub} → {transfer.toClub === "Released" ? <span className="text-red-500">Released</span> : transfer.toClub}
                  </p>
                </div>
                <div className="text-right shrink-0">
                  <div className={`text-sm font-bold ${darkMode ? "text-gray-100" : "text-gray-900"}`}>{transfer.fee}</div>
                  <div className={`text-xs ${darkMode ? "text-gray-500" : "text-gray-400"}`}>{transfer.position}</div>
                </div>
              </div>
            ))}
          </div>
        </div>
      </section>

      {/* Opinion */}

        <div className="flex items-center justify-between mb-5">
          <div className="flex items-center gap-3">
            <div className="w-1 h-6 bg-gray-600 rounded-full" />
            <h2 className={`font-display font-bold text-2xl uppercase tracking-wide ${darkMode ? "text-white" : "text-gray-900"}`}>
              Opinion & Analysis
            </h2>
          </div>
          <Link to="/opinion" className="flex items-center gap-1 text-sm text-gray-500 font-semibold">
            More Opinion <ArrowRight size={14} />
          </Link>
        </div>
        <div className="grid grid-cols-1 lg:grid-cols-2 gap-5">
          {opinionArticles.map((article) => (
            <Link
              key={article.id}
              to={`/article/${article.slug}`}
              className={`group flex gap-4 p-5 rounded-xl border hover:shadow-md transition-all ${darkMode ? "bg-gray-900 border-gray-800" : "bg-white border-gray-100"}`}
            >
              <div className="w-24 h-20 rounded-lg overflow-hidden bg-gray-200 shrink-0">
                <img src={article.thumbnail} alt={article.title} className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-300" />
              </div>
              <div>
                <div className="flex items-center gap-2 mb-2">
                  <span className="text-xs font-bold text-gray-500 uppercase tracking-wider">Opinion</span>
                </div>
                <h3 className={`font-display font-bold text-base leading-snug line-clamp-2 group-hover:text-[#00C853] transition-colors ${darkMode ? "text-gray-100" : "text-gray-900"}`}>
                  {article.title}
                </h3>
                <p className={`text-xs mt-1 ${darkMode ? "text-gray-400" : "text-gray-500"}`}>
                  {article.author} · {new Date(article.date).toLocaleDateString("en-ZA", { day: "numeric", month: "short" })}
                </p>
              </div>
            </Link>
          ))}
        </div>
      </section>

      {/* Video Section */}
      <section className={`border-y py-8 ${darkMode ? "border-gray-800 bg-gray-950" : "border-gray-100 bg-gray-50"}`}>
        <div className="max-w-[1440px] mx-auto px-4">
          <div className="flex items-center justify-between mb-5">
            <div className="flex items-center gap-3">
              <Play size={18} className="text-[#00C853]" />
              <h2 className={`font-display font-bold text-2xl uppercase tracking-wide ${darkMode ? "text-white" : "text-gray-900"}`}>
                Videos
              </h2>
            </div>
            <Link to="/videos" className="flex items-center gap-1 text-sm text-[#00C853] font-semibold">
              All Videos <ArrowRight size={14} />
            </Link>
          </div>
          <div className="grid grid-cols-1 lg:grid-cols-3 gap-5">
            <div className="lg:col-span-1">
              <VideoCard video={featuredVideo} darkMode={darkMode} large />
            </div>
            <div className="lg:col-span-2 grid grid-cols-1 sm:grid-cols-3 gap-4">
              {latestVideos.map((v) => (
                <VideoCard key={v.id} video={v} darkMode={darkMode} />
              ))}
            </div>
          </div>
        </div>
      </section>

      {/* Newsletter */}
      <section className="max-w-[1440px] mx-auto px-4 py-12">
        <div className="bg-[#0A0A0A] rounded-2xl p-8 sm:p-12 text-center">
          <div className="flex justify-center mb-4">
            <div className="w-12 h-12 bg-[#00C853]/10 rounded-full flex items-center justify-center">
              <Mail size={20} className="text-[#00C853]" />
            </div>
          </div>
          <h2 className="font-display font-bold text-3xl text-white mb-2">
            Never Miss a Kick
          </h2>
          <p className="text-gray-400 text-sm mb-6 max-w-md mx-auto">
            Get the best South African football stories delivered to your inbox every morning. Join 28,940 subscribers.

            <div className="bg-[#00C853]/10 border border-[#00C853]/30 text-[#00C853] rounded-xl py-3 px-6 inline-block text-sm font-semibold">
              ✓ You're subscribed! Welcome to DiskiCentral.

          ) : (
            <form onSubmit={handleSubscribe} className="flex flex-col sm:flex-row gap-3 max-w-md mx-auto">

                type="email"
                value={email}
                onChange={(e) => setEmail(e.target.value)}
                placeholder="Enter your email address"
                className="flex-1 px-4 py-3 bg-gray-800 border border-gray-700 text-white rounded-xl text-sm outline-none focus:border-[#00C853] transition-colors placeholder-gray-500"

                type="submit"
                className="px-6 py-3 bg-[#00C853] text-black font-bold text-sm rounded-xl hover:bg-[#00A344] transition-colors"

            </form>
          )}

        </div>

}
